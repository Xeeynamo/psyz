cmake_minimum_required(VERSION 3.16)
project(ztest C)

add_library(ztest STATIC ztest.c)
target_include_directories(ztest PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(ztest PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
if(MSVC)
    target_compile_options(ztest PUBLIC /Zc:preprocessor)
endif()
if(PSP)
    target_link_libraries(ztest PUBLIC pspdisplay)
endif()

include(CheckSymbolExists)
foreach(fn memcpy memmove memset memcmp strlen strcmp strncmp strchr strcspn
           strcpy strcat abs)
    string(TOUPPER ${fn} FN)
    if(fn STREQUAL "abs")
        check_symbol_exists(${fn} "stdlib.h" ZTEST_HAVE_${FN})
    else()
        check_symbol_exists(${fn} "string.h" ZTEST_HAVE_${FN})
    endif()
    if(NOT ZTEST_HAVE_${FN})
        target_compile_definitions(ztest PRIVATE ZTEST_NEED_${FN})
    endif()
endforeach()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_executable(ztest_selftest selftest/selftest.c)
    target_link_libraries(ztest_selftest PRIVATE ztest)
    set_target_properties(ztest_selftest PROPERTIES C_STANDARD 11)
    if(NOT MSVC)
        target_compile_options(ztest_selftest PRIVATE -Wall -Wextra)
        target_compile_options(ztest PRIVATE -Wall -Wextra)
    endif()
    if(NINTENDO_DS)
        file(COPY ${CMAKE_CURRENT_SOURCE_DIR}/selftest/expected
             DESTINATION ${CMAKE_CURRENT_BINARY_DIR}/nitrofs
             PATTERN "*.actual.png" EXCLUDE)
        nds_create_rom(ztest_selftest NAME "ztest selftest"
                       NITROFS ${CMAKE_CURRENT_BINARY_DIR}/nitrofs)
    endif()
    if(PSP)
        create_pbp_file(TARGET ztest_selftest TITLE "ztest selftest")
    endif()
endif()
